Private Sub AccuracySet(ByVal key As String, ByVal value As Variant)
    Dim ws As Worksheet, row As Long
    Set ws = RPX_Sheet("設定")
    For row = 4 To 1000
        If CStr(ws.cells(row, 4).Value2) = key Then ws.cells(row, 2).Value2 = value: Exit Sub
    Next row
    err.Raise 5, "RPX_UI", "ACCURACY_SETTING_MISSING " & key
End Sub
Public Sub RPX_AccuracyPreset()
    If RPX_Busy Then err.Raise 5, "RPX_UI", "ANALYSIS_IS_RUNNING"
    RPX_InternalWrite = True
    RPX_EnsureG1R18Settings
    AccuracySet "MODE", "Fs": AccuracySet "TARGET", 1000: AccuracySet "QUADRATURE", 8: AccuracySet "FS_TOL", 0.0001: AccuracySet "GAP", 0.01
    AccuracySet "ADAPT", 1: AccuracySet "CYCLES", 8: AccuracySet "RAW_LOWER_AUDIT", 1: AccuracySet "C0_AUTO", 0
    AccuracySet "ADAPT_ACCURACY_CAP", 24000: AccuracySet "ADAPT_ACCURACY_ROUNDS", 8
    AccuracySet "ADAPT_LOWER_TARGET", 4000: AccuracySet "ADAPT_LOWER_FAN_ROUNDS", 1: AccuracySet "ADAPT_CERT_QUADRATURE", 8
    AccuracySet "ACCURACY_DLL_RECOVERY", 1: AccuracySet "ACCURACY_DLL_SECONDS", 120: AccuracySet "ACCURACY_DLL_DUAL_SECONDS", 20
    AccuracySet "ACCURACY_DLL_CONE_SCALING", 1: AccuracySet "HSD_SOLVE_SECONDS", 300
    RPX_InternalWrite = False: RPX_Invalidate
End Sub
